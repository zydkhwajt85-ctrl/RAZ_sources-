local cameraFixed = false

local function instantReset()
	local char = LP.Character
	local humanoid = char and char:FindFirstChild("Humanoid")
	local rootPart = char and char:FindFirstChild("HumanoidRootPart")

	if humanoid then
		humanoid.WalkSpeed = 16
	end
	if rootPart then
		rootPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
	end

	local carpet = char and char:FindFirstChild("Flying Carpet")
	if carpet then
		carpet:Destroy()
	end

	local camera = workspace.CurrentCamera
	camera.CameraType = Enum.CameraType.Scriptable
	camera.CFrame = camera.CFrame

	if char then
		for _, part in ipairs(char:GetDescendants()) do
			if part:IsA("BasePart") then
				part.CanCollide = false
			end
		end
	end

	if humanoid then
		humanoid.Health = 0
	end
	if rootPart then
		rootPart.Velocity = Vector3.new(0, 0, 0)
		rootPart.CFrame = CFrame.new(0, 999999, 0)
	end
	if char then
		char:BreakJoints()
	end

	if not cameraFixed then
		cameraFixed = true
		LP.CharacterAdded:Connect(function(newChar)
			task.wait(0.1)
			local newHumanoid = newChar:WaitForChild("Humanoid")
			local currentCamera = workspace.CurrentCamera
			currentCamera.CameraSubject = newHumanoid
			currentCamera.CameraType = Enum.CameraType.Custom
		end)
	end
end

	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then

	end

-- =================== LOGIC ===================
local function saveConfig()
	if writefile then
		pcall(function()
			writefile("hanami_insta_reset.json", HS:JSONEncode(config))
		end)
	end
end

local function updateMode()
	if config.mode == "pc" then
		pcFrame.Visible = true
		mobileFrame.Visible = false
		modeBtn.Text = "MOBILE"
	else
		pcFrame.Visible = false
		mobileFrame.Visible = true
		modeBtn.Text = "PC"
	end
end

modeBtn.MouseButton1Click:Connect(function()
	config.mode = (config.mode == "pc") and "mobile" or "pc"
	updateMode()
	saveConfig()
end)

-- Keybind changing
keybindBtn.MouseButton1Click:Connect(function()
	if listening then return end
	listening = true
	keybindBtn.Text = "..."

	local conn
	conn = UIS.InputBegan:Connect(function(input, gpe)
		if gpe then return end
		if input.UserInputType == Enum.UserInputType.Keyboard then
			if input.KeyCode == Enum.KeyCode.Escape then
				keybindBtn.Text = config.keybind
			else
				config.keybind = input.KeyCode.Name
				currentKey = input.KeyCode
				keybindBtn.Text = config.keybind
				saveConfig()
			end
			listening = false
			if conn then conn:Disconnect() end
		end
	end)
end)

-- PC keybind trigger
UIS.InputBegan:Connect(function(input, gpe)
	if gpe or listening then return end
	if config.mode ~= "pc" then return end
	if input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode == currentKey then
		instantReset()
	end
end)

-- Mobile button trigger
resetBtn.MouseButton1Click:Connect(function()
	local pressTween = TS:Create(resetBtn, TweenInfo.new(0.05), {
		BackgroundColor3 = Color3.fromRGB(60, 25, 25)
	})
	pressTween:Play()

	instantReset()

	task.delay(0.1, function()
		local releaseTween = TS:Create(resetBtn, TweenInfo.new(0.1), {
			BackgroundColor3 = Color3.fromRGB(35, 15, 15)
		})
		releaseTween:Play()
	end)
end)

-- =================== LOAD ===================
local function loadConfig()
	local loaded = false
	if isfile and isfile("hanami_insta_reset.json") then
		pcall(function()
			local data = HS:JSONDecode(readfile("hanami_insta_reset.json"))
			if data then
				if data.keybind then
					config.keybind = data.keybind
					local key = Enum.KeyCode[data.keybind]
					if key then
						currentKey = key
						keybindBtn.Text = data.keybind
					end
				end
				if data.mode == "pc" or data.mode == "mobile" then
					config.mode = data.mode
					loaded = true
				end
			end
		end)
	end

	-- Auto-detect mobile if no saved config
	if not loaded and UIS.TouchEnabled and not UIS.KeyboardEnabled then
		config.mode = "mobile"
	end

	updateMode()
end

loadConfig()
